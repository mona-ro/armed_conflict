Comparison of Week 3 and Independently Reconstructed Final Data

Purpose

This report compares the Week 3 final dataset with an independent reconstruction. The reconstruction is created from raw files by R/create_final_data_agent.R; it never reads final_data.csv. Run the reconstruction first, then render this document from the repository root.

library(readr)
library(dplyr)
library(tibble)
library(knitr)

week3_path <- here::here("data", "processed", "final_data.csv")
agent_path <- here::here("data", "processed", "final_data_agent.csv")
if (!file.exists(week3_path)) {
  stop("Week 3 dataset not found: ", week3_path)
}
if (!file.exists(agent_path)) {
  stop("Independent dataset not found: ", agent_path,
       ". Run R/create_final_data_agent.R from the repository root first.")
}

week3 <- read_csv(week3_path, show_col_types = FALSE)
agent <- read_csv(agent_path, show_col_types = FALSE)
key <- c("iso", "year")
type_name <- function(x) paste(class(x), collapse = "/")
same_value <- function(x, y) {
  if (is.numeric(x) && is.numeric(y)) {
    (is.na(x) & is.na(y)) | (!is.na(x) & !is.na(y) & abs(x - y) <= 1e-12)
  } else {
    (is.na(x) & is.na(y)) | (!is.na(x) & !is.na(y) & x == y)
  }
}

Rows are aligned by iso and year rather than their file positions. Only then are values compared. Numeric values use a tolerance of (10^{-12}) to avoid classifying CSV floating-point printing as a substantive mismatch.

Row-level values

common <- intersect(names(week3), names(agent))
variables <- setdiff(common, key)
w <- arrange(week3, across(all_of(key)))
a <- arrange(agent, across(all_of(key)))
keys_align <- nrow(only_week3) == 0 && nrow(only_agent) == 0 &&
  nrow(week3) == nrow(week3_keys) && nrow(agent) == nrow(agent_keys)
columns_align <- setequal(names(week3), names(agent))

if (!keys_align || !columns_align) {
  variable_differences <- tibble(variable = character(), differing_observations = integer())
  details <- tibble()
  cat("Row-level comparison was not run because keys or columns do not align.\n")
} else {
  variable_differences <- bind_rows(lapply(variables, function(v) {
    tibble(variable = v, differing_observations = sum(!same_value(w[[v]], a[[v]])))
  }))
  kable(filter(variable_differences, differing_observations > 0))

  details <- bind_rows(lapply(variables, function(v) {
    equal <- same_value(w[[v]], a[[v]])
    tibble(iso = w$iso[!equal], year = w$year[!equal], variable = v,
           week3_value = as.character(w[[v]][!equal]),
           agent_value = as.character(a[[v]][!equal]))
  }))
  if (nrow(details) == 0) {
    cat("No row-level differences were found in common variables.\n")
  } else {
    kable(details)
  }
}

Interpretation and raw-data logic

all_schema <- identical(names(week3), names(agent)) && all(schema$match)
all_missing <- all(missingness$match)
all_keys <- nrow(only_week3) == 0 && nrow(only_agent) == 0 &&
  nrow(week3) == nrow(week3_keys) && nrow(agent) == nrow(agent_keys)
all_values <- sum(variable_differences$differing_observations) == 0

if (all_schema && all_missing && all_keys && all_values) {
  cat("The datasets agree on dimensions, schema, missingness, country-year keys, and all values. No correction is supported by this evidence.\n")
} else {
  cat("The datasets do not fully agree. The tables identify every differing variable and country-year. Inspect those raw records before changing the Week 3 file.\n")
}
